Selecting Elements
Every way to grab elements from the DOM
Modern Selectors (Use These)
querySelector
// Returns FIRST match — CSS selector syntax
const btn = document.querySelector('button');
const box = document.querySelector('#myId');
const item = document.querySelector('.active');
const link = document.querySelector('nav a');
const inp = document.querySelector('input[type="email"]');
const nth = document.querySelector('li:nth-child(2)');
// Returns null if not found
querySelectorAll
// Returns NodeList — ALL matches
const all = document.querySelectorAll('li');
const btns = document.querySelectorAll('.btn');

// NodeList is NOT an array — convert with:
const arr = [...all];          // spread
const arr2 = Array.from(all); // Array.from

// forEach works directly on NodeList
all.forEach(el => el.style.color = 'red');

Navigating the tree — parents, children, siblings
Parent Navigation
parentNode vs parentElement
const el = document.querySelector('li');

el.parentNode     // parent — any node type
el.parentElement  // parent — only Elements

// Difference matters at root:
document.documentElement.parentNode    // #document
document.documentElement.parentElement // null

// Walk up chain:
el.parentElement.parentElement
closest() for Ancestors
// Find nearest ancestor matching selector
const card = el.closest('.card');
const section = el.closest('section');
const form = el.closest('form');

// Includes the element itself!
const p = p.closest('p'); // returns p itself

// Returns null if not found
if(el.closest('.modal')) { /* inside modal */ }
Children Navigation
children vs childNodes
const ul = document.querySelector('ul');

// Element children only (HTMLCollection)
ul.children         // live HTMLCollection
ul.children[0]      // first element child
ul.childElementCount // count

// All nodes incl. text, comments (NodeList)
ul.childNodes        // includes whitespace text!

// First/Last element shortcuts
ul.firstElementChild  // ✅ use this
ul.lastElementChild   // ✅ use this
ul.firstChild         // ⚠️ may be text node
Iterating Children
const ul = document.querySelector('ul');

// Best: spread children into array
[...ul.children].forEach(li => {
  console.log(li.textContent);
});

// Or use for...of directly
for(const li of ul.children) {
  li.style.color = 'blue';
}

// Array methods on children
const texts = [...ul.children].map(li => li.textContent);
Sibling Navigation
nextSibling vs nextElementSibling
const el = document.querySelector('li.active');

// ✅ Use Element versions (skip text nodes)

el.previousElementSibling   // previous element

// ⚠️ Node versions include text/comments
el.nextSibling      // could be text node
el.previousSibling  // could be text node

let sib = el.nextElementSibling;

  console.log(sib);
  sib = sib.nextElementSibling;

Creating Nodes
createElement
// Create any HTML element
const div = document.createElement('div');

const img = document.createElement('img');

// Build it up before inserting

Real-World Pattern
// Complete create → configure → insert
function createCard(title, body) {
  const card = document.createElement('div');
  card.className = 'card';

  const h2 = document.createElement('h2');
  h2.textContent = title;

  const p = document.createElement('p');
  p.textContent = body;

  card.append(h2, p);  // append multiple
  return card;
}
document.body.append(createCard('Hi', 'World'));
Inserting Elements
append / prepend (Modern ✅)
const ul = document.querySelector('ul');
const li = document.createElement('li');
li.textContent = 'New item';

// append() — adds to END, accepts multiple
ul.append(li);                    // element
ul.append('text');               // text string!
ul.append(li1, li2, 'text');    // multiple args

// prepend() — adds to START
ul.prepend(li);

// before() / after() — relative to element
li.before(newEl);   // insert before li
li.after(newEl);    // insert after li
appendChild / insertBefore (Legacy)
// appendChild — adds to end, one node only
parent.appendChild(child);

// insertBefore(newNode, referenceNode)
parent.insertBefore(newEl, existingEl);

// Insert at end with insertBefore
parent.insertBefore(newEl, null);

// vs append() — returns nothing
// append() returns undefined
// appendChild() returns appended node
const appended = parent.appendChild(node);
insertAdjacentHTML / Element / Text
4 Insert Positions
/* Position strings:
   "beforebegin"  — before the element itself
   "afterbegin"   — inside element, before first child
   "beforeend"    — inside element, after last child
   "afterend"     — after the element itself        */

el.insertAdjacentHTML('beforebegin', '<p>Before</p>');
el.insertAdjacentHTML('afterbegin',  '<span>First inside</span>');
el.insertAdjacentHTML('beforeend',   '<span>Last inside</span>');
el.insertAdjacentHTML('afterend',    '<p>After</p>');

// Same positions for elements and text:
el.insertAdjacentElement('afterend', newEl);
el.insertAdjacentText('beforeend', 'safe text');

/* insertAdjacentHTML is FAST — doesn't re-parse existing DOM */
⚠️ insertAdjacentHTML with user input = XSS risk. Sanitize first or use insertAdjacentText.
DocumentFragment — Performance Pattern
Batch Insert with Fragment
// Without fragment — 100 DOM updates (slow!)
const ul = document.querySelector('ul');
data.forEach(item => {
  const li = document.createElement('li');
  li.textContent = item;
  ul.appendChild(li);  // ← repaints each time
});

// With fragment — 1 DOM update (fast! ✅)
const frag = document.createDocumentFragment();
data.forEach(item => {
  const li = document.createElement('li');
  li.textContent = item;
  frag.appendChild(li);  // ← in memory only
});
ul.appendChild(frag);    // ← ONE repaint
✅ Use DocumentFragment when inserting multiple elements in a loop — massive performance win on large lists.

innerHTML, textContent, innerText and more
The Big Three
innerHTML
// READ: gets HTML string including tags
div.innerHTML   // "<b>Hello</b>"

// WRITE: parses and renders HTML
div.innerHTML = '<b>Bold</b> text';
div.innerHTML = '';  // clear content

// Template literal pattern:
div.innerHTML = `
  <h2>${user.name}</h2>
  <p>${user.bio}</p>
`;
🚨 XSS Risk: Never put user input directly into innerHTML. Use textContent for user data.
textContent vs innerText
// textContent — raw text, ignores CSS
el.textContent          // returns all text
el.textContent = 'Hi'  // safe from XSS ✅

// innerText — aware of CSS visibility
el.innerText            // respects display:none
el.innerText = 'Hi'    // triggers layout
textContent innerText
Hidden els ✅ Included ❌ Excluded
Performance ⚡ Fast 🐢 Slow
Parses HTML ❌ No ❌ No
Use when Most cases Screen text

HTML attributes, properties, and dataset
Attribute Methods
get / set / remove / has
const el = document.querySelector('img');

// Get attribute value (string or null)
el.getAttribute('src')         // → "/img.png"
el.getAttribute('alt')         // → "text"

// Set attribute
el.setAttribute('src', '/new.png');

el.removeAttribute('disabled');

// Check attribute exists
el.hasAttribute('hidden')   // true/false

// Get all attributes
el.attributes  // NamedNodeMap
Attr vs Property
// Attributes: HTML string representation
// Properties: JS live value

// Example:
<input type="text" value="initial">
input.getAttribute('value')  // "initial" (always)
input.value                  // current typed value

// Boolean attr: presence = true
<button disabled>
btn.hasAttribute('disabled') // true
btn.disabled                 // true (property)

// Reflect pattern: use properties for live state
checkbox.checked = true;   // ✅ use property
dataset (data-* attributes)
Reading & Writing dataset
// HTML: data-user-id="42" data-is-active="true"
const el = document.querySelector('[data-user-id]');

// Read — camelCase conversion auto
el.dataset.userId    // "42" (always string!)
el.dataset.isActive  // "true"

// Write
el.dataset.userId = 99;
el.dataset.newProp = 'hello';
// → adds data-new-prop="hello"

// Delete
delete el.dataset.userId;

// Query by dataset
document.querySelector('[data-user-id="42"]');

// Pass data to event handlers via dataset
<button data-id="5" data-action="delete">Del</button>

document.addEventListener('click', e => {
  const { id, action } = e.target.dataset;
  if(action === 'delete') {
    deleteItem(Number(id));
  }
});

// State management pattern
card.dataset.expanded = 'false';
const isOpen = card.dataset.expanded === 'true';

Events
The complete event system — listeners, propagation, delegation
addEventListener
Core Syntax
// addEventListener(type, handler, options)
el.addEventListener('click', handler);

// Options object (3rd argument)
el.addEventListener('click', handler, {
  once:    true,   // auto-remove after first fire
  capture: false,  // bubble phase (default)
  passive: true,   // can't call preventDefault (perf!)
});

// Remove listener (must be SAME reference!)
function handler(e) { /* ... */ }
el.addEventListener('click', handler);
el.removeEventListener('click', handler); // ✅ works

// ❌ Arrow function in both — WON'T remove!
el.addEventListener('click', () => {});
el.removeEventListener('click', () => {}); // ❌
Event Object (e)
el.addEventListener('click', function(e) {
  e.target          // element clicked
  e.currentTarget   // element with listener
  e.type            // "click"
  e.timeStamp       // ms since page load
  e.bubbles         // does it bubble?
  e.cancelable      // can be cancelled?

  // Stop propagation
  e.stopPropagation();       // stop bubbling
  e.stopImmediatePropagation(); // stop all handlers

  // Prevent default behavior
  e.preventDefault();       // e.g. stop form submit
});
Event Propagation
Bubbling & Capturing
/* Event flow:
 * 1. CAPTURE phase: window → document → body → el
 * 2. TARGET phase: at the element itself
 * 3. BUBBLE phase: el → body → document → window */

// Capture phase listener
parent.addEventListener('click', fn, true);

// Bubble phase (default)
parent.addEventListener('click', fn, false);

// Stop at element, don't bubble up
btn.addEventListener('click', e => {
  e.stopPropagation();
  // parent won't hear this click
});
Event Delegation ⭐ KEY PATTERN
// Instead of 100 listeners (bad):
items.forEach(i => i.addEventListener('click', fn));

// Use ONE listener on parent (good ✅)
document.querySelector('ul')
  .addEventListener('click', e => {
    if(e.target.matches('li')) {
      console.log(e.target.textContent);
    }
    if(e.target.matches('.delete-btn')) {
      e.target.closest('li').remove();
    }
  });
✅ Delegation works for dynamically added elements too!

Advanced DOM
MutationObserver, IntersectionObserver, Templates, Storage
MutationObserver
Watch DOM Changes
const observer = new MutationObserver(mutations => {
  mutations.forEach(m => {
    console.log(m.type);          // "childList","attributes","characterData"
    console.log(m.addedNodes);    // NodeList of added
    console.log(m.removedNodes);  // NodeList of removed
    console.log(m.attributeName); // changed attr name
    console.log(m.oldValue);      // previous value
  });
});

observer.observe(el, {
  childList:     true,  // watch add/remove of children
  attributes:    true,  // watch attribute changes
  subtree:       true,  // watch all descendants
  characterData: true,  // watch text content changes
  attributeOldValue: true,   // get old attr value
});

observer.disconnect(); // stop watching
IntersectionObserver
Visibility / Lazy Loading
// Detect when elements enter/leave viewport
const io = new IntersectionObserver((entries) => {
  entries.forEach(entry => {
    entry.isIntersecting   // true = visible
    entry.intersectionRatio // 0 to 1
    entry.target           // observed element

    if(entry.isIntersecting) {
      entry.target.classList.add('visible'); // animate in
      io.unobserve(entry.target);             // stop watching
    }
  });
}, {
  threshold: 0.2,    // 20% visible to trigger
  rootMargin: '0px 0px -50px 0px'  // shrink viewport
});

document.querySelectorAll('.animate-on-scroll')
         .forEach(el => io.observe(el));

// Lazy load images
const lazyIO = new IntersectionObserver(entries => {
  entries.forEach(e => {
    if(e.isIntersecting) {
      e.target.src = e.target.dataset.src;
      lazyIO.unobserve(e.target);
    }
  });
});
document.querySelectorAll('img[data-src]').forEach(img => lazyIO.observe(img));

DOM Interview Questions
10 questions that actually get asked — with full answers
Q1
What is the difference between innerHTML, textContent, and innerText? When would you use each?
EasyContent
▼
innerHTML — Parses and renders HTML markup. Can set/get HTML including tags. Security risk: never use with user input (XSS). Use for injecting trusted HTML templates.

textContent — Gets/sets raw text only, ignores HTML tags. Includes hidden elements. Fastest option and XSS-safe. Use for setting user-generated content.

innerText — Like textContent but CSS-aware: respects display:none and visibility:hidden. Triggers layout recalculation (slower). Use when you need what the user visually sees. 
// Rule of thumb:
el.innerHTML = trustedHTML;     // ✅ for HTML structure
el.textContent = userInput;    // ✅ always safe for user data
el.innerText;                  // ✅ when CSS-aware read is needed
Q2
Explain Event Bubbling, Capturing, and how stopPropagation() differs from stopImmediatePropagation().
MediumEvents
▼
Events travel in 3 phases:
1. Capture: window → document → body → target (top-down)
2. Target: event fires on the actual element
3. Bubble: target → body → document → window (bottom-up)

stopPropagation() — stops the event from traveling further in the current direction, but other listeners on the same element still fire.
stopImmediatePropagation() — stops propagation AND prevents any other handlers on the same element from running. 
// Multiple listeners on same button
btn.addEventListener('click', e => {
  e.stopImmediatePropagation();
  console.log('1st — runs');
});
btn.addEventListener('click', () => console.log('2nd — NEVER runs'));
Q3
What is Event Delegation? Why is it better than attaching listeners to each element? Write an example.
MediumEventsPerformance
▼
Event delegation is attaching a single listener to a parent and using e.target to determine which child triggered it. Benefits: uses one listener instead of many (memory efficient), works for dynamically added elements, simpler cleanup. 
// ❌ Bad: 1000 listeners for 1000 list items
document.querySelectorAll('li').forEach(li =>
  li.addEventListener('click', handleClick)
);

// ✅ Good: ONE listener handles all
document.querySelector('ul').addEventListener('click', e => {
  const li = e.target.closest('li'); // handle nested elements
  if(!li) return;
  handleClick(li);
});
Q4
What is the difference between e.target and e.currentTarget?
EasyEvents
▼
e.target — the element that actually triggered the event (where the user clicked).

e.currentTarget — the element the event listener is attached to. During bubbling this is the ancestor.

Inside the handler, this equals e.currentTarget for regular functions (not arrow functions). 
// Click on <span> inside <button>
btn.addEventListener('click', e => {
  e.target         // → <span> (where click landed)
  e.currentTarget  // → <button> (has the listener)
});
Q5
What is the difference between querySelector returning a static NodeList vs getElementsByClassName returning a live HTMLCollection?
MediumSelectors
▼
A live collection (HTMLCollection from getElementsBy*) automatically updates when the DOM changes. A static NodeList (from querySelectorAll) is a snapshot taken at query time and never updates. 
const live = document.getElementsByClassName('item'); // live
const snap = document.querySelectorAll('.item');      // static

console.log(live.length); // 2
console.log(snap.length); // 2

document.body.insertAdjacentHTML('beforeend', '<div class="item"></div>');

console.log(live.length); // 3 ← auto updated
console.log(snap.length); // 2 ← still 2
Q6
Explain the difference between HTML attributes and DOM properties. Give a concrete example.
MediumAttributes
▼
Attributes are what's in the HTML source — always strings, represent the initial value.
Properties are live JS object properties — can be any type, represent the current value.

They start in sync but diverge once the user or JS changes the value. 
// HTML: <input type="text" value="initial">
// User types "hello"
input.getAttribute('value')  // "initial" — never changes
input.value                  // "hello" — current live value

// Checkbox: attribute vs property
// HTML: <input type="checkbox" checked>
input.getAttribute('checked')  // "" (empty string — presence)
input.checked                  // true (boolean property)
input.removeAttribute('checked');    // removes from HTML
input.checked = false;          // changes live state ✅
Q7
Why is DocumentFragment better for inserting multiple elements? How does it improve performance?
MediumPerformance
▼
Every DOM insertion causes a reflow/repaint — the browser recalculates layout. Inserting 100 elements individually causes 100 reflows. A DocumentFragment exists in memory (off-screen), so you can add all elements to it without any reflow. When you append the fragment to the DOM, it causes exactly one reflow. 
// DocumentFragment — batch 1000 items in ONE reflow
const frag = document.createDocumentFragment();
for(let i = 0; i < 1000; i++) {
  const li = document.createElement('li');
  li.textContent = `Item ${i}`;
  frag.append(li);    // ← no DOM reflow
}
document.querySelector('ul').append(frag); // ← ONE reflow
Q8
What is MutationObserver and when would you use it instead of event listeners?
HardAdvanced
▼
MutationObserver watches for DOM changes (added/removed nodes, attribute changes, text changes) that you didn't cause — e.g., changes made by third-party scripts, frameworks, or dynamic content loaders. Use it when events won't tell you about DOM changes, like when a chat SDK injects a widget, when you need to react to a library modifying the DOM, or implementing undo/redo history. 
// Watch a container for injected content
const obs = new MutationObserver(muts => {
  muts.forEach(m => {
    m.addedNodes.forEach(node => {
      if(node.nodeType === 1) {  // ELEMENT_NODE
        console.log('New element added:', node);
      }
    });
  });
});
obs.observe(document.body, { childList: true, subtree: true });
Q9
How does getBoundingClientRect() work? How do you check if an element is visible in the viewport without IntersectionObserver?
HardLayout
▼
getBoundingClientRect() returns a DOMRect with the element's size and position relative to the current viewport. Positions are not affected by scrolling (it's viewport-relative). It causes a synchronous layout/reflow when called, so avoid it in tight loops. 
function isInViewport(el) {
  const rect = el.getBoundingClientRect();
  return(
    rect.top    >= 0 &&
    rect.left   >= 0 &&
    rect.bottom <= (window.innerHeight) &&
    rect.right  <= (window.innerWidth)
  );
}
// More efficient alternative: use IntersectionObserver!
Q10
How would you implement a basic pub/sub (event bus) using custom DOM events?
HardPatternsAdvanced
▼
The DOM's event system is itself a pub/sub. Using a shared EventTarget (like document) as an event bus lets loosely coupled modules communicate without direct references. 
// EventBus using document as shared target
const bus = {
  on(event, cb) {
    document.addEventListener(event, e => cb(e.detail));
  },
  emit(event, data) {
    document.dispatchEvent(new CustomEvent(event, { detail: data }));
  }
};

// Module A subscribes
bus.on('cart:update', data => console.log('Cart updated:', data));

// Module B publishes
bus.emit('cart:update', { items: 3, total: 99.99 });

// Output: [0,1]  (nums[0]+nums[1]=9)
function twoSum(nums, target) {
  const map = new Map();
  for(let i = 0; i < nums.length; i++) {
    const complement = target - nums[i];
    if(map.has(complement)) return [map.get(complement), i];

  }

Maximum Subarray Sum (Kadane's)

// Output: 6 (subarray [4,-1,2,1])
function maxSubArray(nums) {
  let max = nums[0], curr = nums[0];
  for(let i = 1; i < nums.length; i++) {

function deepMerge(target, source) {
  const result = { ...target };
  for(const key in source) {
    if(source[key] && typeof source[key] === 'object'
    && !Array.isArray(source[key])) {
      result[key] = deepMerge(target[key] || {}, source[key]);
    } else {
      result[key] = source[key];
    }
  }
  return result;
}

Debounce Function

function debounce(fn, wait) {
  let timer;
  return function(...args) {
    clearTimeout(timer);
    timer = setTimeout(() => {
      fn.apply(this, args);
    }, wait);
  };
}
